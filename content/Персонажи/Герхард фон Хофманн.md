<script>
(function() {
    // Здесь вы можете менять ранги (от A до E) для каждого персонажа
    const config = {
        stats: [
            { name: "СИЛА", value: "B" },
            { name: "ИНТЕЛЛЕКТ", value: "B" },
            { name: "МАГИЯ", value: "C" },
            { name: "ВЫНОСЛИВОСТЬ", value: "A" },
            { name: "ТОЧНОСТЬ", value: "D" },
            { name: "ПОТЕНЦИАЛ", value: "C" }
        ],
        ranks: ["A", "B", "C", "D", "E"],
        size: 300,
        outerRadius: 130,
        radarRadius: 90
    };

    const rankToValue = { "A": 100, "B": 80, "C": 60, "D": 40, "E": 20 };
    const svg = document.getElementById('radar-svg');
    if (!svg) return;
    
    const outerRimGroup = document.getElementById('outerRim');
    const statTextsGroup = document.getElementById('statTexts');
    const radarChartGroup = document.getElementById('radarChart');
    
    outerRimGroup.innerHTML = '';
    statTextsGroup.innerHTML = '';
    radarChartGroup.innerHTML = '';

    const numStats = config.stats.length;
    const centerX = config.size / 2;
    const centerY = config.size / 2;

    for (let i = 0; i < 48; i++) {
        const angle = (i * 360) / 48;
        const radians = (angle * Math.PI) / 180;
        const isMajor = i % 2 === 0;
        const tickLen = isMajor ? 10 : 5;
        const r1 = config.outerRadius;
        const r2 = r1 - tickLen;
        const x1 = centerX + r1 * Math.cos(radians - Math.PI/2);
        const y1 = centerY + r1 * Math.sin(radians - Math.PI/2);
        const x2 = centerX + r2 * Math.cos(radians - Math.PI/2);
        const y2 = centerY + r2 * Math.sin(radians - Math.PI/2);
        outerRimGroup.innerHTML += `<line x1="${x1}" y1="${y1}" x2="${x2}" y2="${y2}" stroke="currentColor" stroke-width="${isMajor ? 2 : 1}" />`;
    }

    config.stats.forEach((stat, i) => {
        const angle = (i * 360) / numStats;
        const radians = (angle * Math.PI) / 180;
        const r = config.radarRadius + 22;
        const x = centerX + r * Math.cos(radians - Math.PI/2);
        const y = centerY + r * Math.sin(radians - Math.PI/2);
        statTextsGroup.innerHTML += `<text x="${x}" y="${y}" fill="currentColor" font-size="11" font-weight="bold" text-anchor="middle" dominant-baseline="middle" transform="rotate(${angle} ${x} ${y})">${stat.name}</text>`;
    });

    config.stats.forEach((_, i) => {
        const radians = ((i * 360) / numStats * Math.PI) / 180;
        const x = centerX + config.radarRadius * Math.cos(radians - Math.PI / 2);
        const y = centerY + config.radarRadius * Math.sin(radians - Math.PI / 2);
        radarChartGroup.innerHTML += `<line x1="${centerX}" y1="${centerY}" x2="${x}" y2="${y}" stroke="currentColor" stroke-opacity="0.3" stroke-width="1" />`;
    });

    config.ranks.forEach((rank) => {
        const r = (rankToValue[rank] / 100) * config.radarRadius;
        radarChartGroup.innerHTML += `<circle cx="${centerX}" cy="${centerY}" r="${r}" stroke="currentColor" stroke-opacity="0.2" fill="none" />`;
        radarChartGroup.innerHTML += `<text x="${centerX + 4}" y="${centerY - r}" fill="currentColor" font-size="10" font-weight="bold" dominant-baseline="middle">${rank}</text>`;
    });

    const points = config.stats.map((stat, i) => {
        const radians = ((i * 360) / numStats * Math.PI) / 180;
        const r = (rankToValue[stat.value] / 100) * config.radarRadius;
        return `${centerX + r * Math.cos(radians - Math.PI / 2)},${centerY + r * Math.sin(radians - Math.PI / 2)}`;
    }).join(" ");

    radarChartGroup.innerHTML += `<polygon points="${points}" fill="rgba(244, 67, 54, 0.6)" stroke="#f44336" stroke-width="2" />`;
})();
</script>
